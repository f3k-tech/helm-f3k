{{- /*
Render access information per enabled protocol (hls, webRtc, rtsp, rtmp, srt, srpt).
Shows Ingress/HTTPRoute URLs and Service access/port-forward hints.
*/ -}}
{{- $protocols := list "hls" "webRtc" "rtsp" "rtmp" "srt" "srpt" -}}
{{- range $p := $protocols -}}
{{- $cfg := index $.Values $p -}}
{{- if and $cfg $cfg.enabled -}}
Protocol: {{ $p }}

{{- /* Ingress URLs */ -}}
{{- if and (hasKey $cfg "ingress") ($cfg.ingress.enabled) -}}
Ingress:
{{- range $host := $cfg.ingress.hosts }}
  {{- range .paths }}
  http{{ if $cfg.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- end }}

{{- /* Gateway API HTTPRoute info */ -}}
{{- if and (hasKey $cfg "httpRoute") ($cfg.httpRoute.enabled) -}}
Gateway API (HTTPRoute):
  Route Name: {{ include "mediamtx.fullname" $ }}-{{ $p }}
  ParentRefs: {{- range $cfg.httpRoute.parentRefs }} {{ .name }}{{ if .namespace }} (ns={{ .namespace }}){{ end }}{{- end }}
  Hostnames: {{- if $cfg.httpRoute.hostnames }} {{ join ", " $cfg.httpRoute.hostnames }} {{- else }} (from gateway listener) {{- end }}
{{- end }}

Service:
  Name: {{ include "mediamtx.fullname" $ }}-{{ $p }}
  Type: {{ $cfg.service.type }}
  Port: {{ $cfg.service.port }}

{{- if contains "LoadBalancer" $cfg.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP.
  kubectl get --namespace {{ $.Release.Namespace }} svc -w {{ include "mediamtx.fullname" $ }}-{{ $p }}
{{- else if contains "NodePort" $cfg.service.type }}
  # NodePort access
  export NODE_PORT=$(kubectl get --namespace {{ $.Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "mediamtx.fullname" $ }}-{{ $p }})
  export NODE_IP=$(kubectl get nodes --namespace {{ $.Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo {{ if or (eq $p "hls") (eq $p "webRtc") }}http{{ else }}tcp{{ end }}://$NODE_IP:$NODE_PORT
{{- else }}
  # Port-forward (local testing)
  kubectl --namespace {{ $.Release.Namespace }} port-forward svc/{{ include "mediamtx.fullname" $ }}-{{ $p }} {{ $cfg.service.port }}:{{ $cfg.service.port }}
  {{- if or (eq $p "hls") (eq $p "webRtc") }}
  echo Visit http://127.0.0.1:{{ $cfg.service.port }}/
  {{- else if eq $p "rtsp" }}
  echo Connect RTSP to rtsp://127.0.0.1:{{ $cfg.service.port }}
  {{- else if eq $p "rtmp" }}
  echo Connect RTMP to rtmp://127.0.0.1:{{ $cfg.service.port }}
  {{- else if eq $p "srt" }}
  echo Connect SRT (UDP) to srt://127.0.0.1:{{ $cfg.service.port }}
  {{- else if eq $p "srpt" }}
  echo SRTP/ICE (UDP) uses dynamic ports; ensure NAT/ICE config.
  {{- end }}
{{- end }}

---
{{- end -}}
{{- end -}}
